Copy-Paste HTML & JS Slideshow Template

Embed a clean 50-photo auto-preloading slideshow on your own website.

<!-- Lightweight 50 Photo Web Slideshow Structure -->
<div id="slideshow-box" style="position:relative; max-width:800px; margin:0 auto;">
  <img id="current-photo" src="photo1.jpg" style="width:100%; height:auto; border-radius:12px;" alt="Photo Slide" />
  <button onclick="changePhoto(-1)" style="position:absolute; left:10px; top:50%;">&lt;</button>
  <button onclick="changePhoto(1)" style="position:absolute; right:10px; top:50%;">&gt;</button>
</div>

<script>
  // Array of 50 photo URLs
  const photos = Array.from({ length: 50 }, (_, i) => `https://picsum.photos/id/${i + 10}/800/600`);
  let currentIndex = 0;
  let timer = null;

  function showPhoto(index) {
    currentIndex = (index + photos.length) % photos.length;
    document.getElementById('current-photo').src = photos[currentIndex];
    preloadNext();
  }

  function changePhoto(step) {
    showPhoto(currentIndex + step);
  }

  // Preload next image to eliminate visual delay
  function preloadNext() {
    const nextImg = new Image();
    nextImg.src = photos[(currentIndex + 1) % photos.length];
  }

  // Start Autoplay (every 3 seconds)
  function startAutoplay() {
    timer = setInterval(() => changePhoto(1), 3000);
  }

  window.onload = startAutoplay;
</script>

Converting 50 Photos to an MP4 Video File via FFmpeg

If you prefer rendering a single video file from 50 images named img001.jpg to img050.jpg, run this terminal command:

ffmpeg -framerate 1/2 -i img%03d.jpg -c:v libx264 -r 30 -pix_fmt yuv420p output_slideshow.mp4

* This sets each photo duration to 2 seconds (`-framerate 1/2`) and outputs an H.264 MP4 compatible with all browsers.
